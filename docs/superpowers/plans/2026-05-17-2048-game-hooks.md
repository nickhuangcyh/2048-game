**Goal:** Integrate game logic with keyboard and touch event listeners using custom hooks and update the main application component.

**Architecture:** 
- `useGame`: A custom hook using `useReducer` to manage the game state, encapsulating actions like moving, restarting, and changing board size.
- `useMoveListeners`: A custom hook that attaches and detaches event listeners for keyboard (Arrow keys, WASD) and touch events (swipe gestures), calling a callback with the detected direction.
- `App.tsx`: The main component that coordinates the hooks and UI.

**Tech Stack:** React, TypeScript

### Task 1: Create `useGame` Hook

**Files:**
- Create: `src/hooks/useGame.ts`

- [ ] **Step 1: Implement `useGame` hook with `useReducer`**

```typescript
import { useReducer, useEffect } from 'react';
import { initGame, moveTiles } from '../logic/gameLogic';
import type { GameState, Direction } from '../types';

type Action =
  | { type: 'MOVE'; direction: Direction }
  | { type: 'RESTART' }
  | { type: 'CHANGE_SIZE'; size: number };

const LOCAL_STORAGE_KEY = '2048-best-score';

const getInitialBestScore = () => {
  const saved = localStorage.getItem(LOCAL_STORAGE_KEY);
  return saved ? parseInt(saved, 10) : 0;
};

const gameReducer = (state: GameState, action: Action): GameState => {
  switch (action.type) {
    case 'MOVE': {
      if (state.status !== 'playing') return state;
      const { newState, hasMoved } = moveTiles(state, action.direction);
      if (!hasMoved) return state;
      
      if (newState.bestScore > state.bestScore) {
        localStorage.setItem(LOCAL_STORAGE_KEY, newState.bestScore.toString());
      }
      
      return newState;
    }
    case 'RESTART': {
      const bestScore = getInitialBestScore();
      return { ...initGame(state.size), bestScore };
    }
    case 'CHANGE_SIZE': {
      const bestScore = getInitialBestScore();
      return { ...initGame(action.size), bestScore };
    }
    default:
      return state;
  }
};

export const useGame = (initialSize: number) => {
  const [state, dispatch] = useReducer(gameReducer, undefined, () => {
    const bestScore = getInitialBestScore();
    return { ...initGame(initialSize), bestScore };
  });

  const move = (direction: Direction) => dispatch({ type: 'MOVE', direction });
  const restart = () => dispatch({ type: 'RESTART' });
  const changeSize = (size: number) => dispatch({ type: 'CHANGE_SIZE', size });

  return { state, move, restart, changeSize };
};
```

### Task 2: Create `useMoveListeners` Hook

- [ ] **Step 1: Implement `useMoveListeners` hook**

```typescript
import { useEffect, useRef } from 'react';
import type { Direction } from '../types';

export const useMoveListeners = (onMove: (direction: Direction) => void) => {
  const touchStart = useRef<[number, number] | null>(null);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      const keyMap: Record<string, Direction> = {
        ArrowUp: 'UP',
        w: 'UP',
        W: 'UP',
        ArrowDown: 'DOWN',
        s: 'DOWN',
        S: 'DOWN',
        ArrowLeft: 'LEFT',
        a: 'LEFT',
        A: 'LEFT',
        ArrowRight: 'RIGHT',
        d: 'RIGHT',
        D: 'RIGHT',
      };

      const direction = keyMap[e.key];
      if (direction) {
        e.preventDefault();
        onMove(direction);
      }
    };

    const handleTouchStart = (e: TouchEvent) => {
      if (e.touches.length > 1) return;
      touchStart.current = [e.touches[0].clientX, e.touches[0].clientY];
    };

    const handleTouchEnd = (e: TouchEvent) => {
      if (!touchStart.current) return;

      const touchEnd = [e.changedTouches[0].clientX, e.changedTouches[0].clientY];
      const dx = touchEnd[0] - touchStart.current[0];
      const dy = touchEnd[1] - touchStart.current[1];
      
      touchStart.current = null;

      const absDx = Math.abs(dx);
      const absDy = Math.abs(dy);

      if (Math.max(absDx, absDy) < 20) return; // Ignore small movements

      if (absDx > absDy) {
        onMove(dx > 0 ? 'RIGHT' : 'LEFT');
      } else {
        onMove(dy > 0 ? 'DOWN' : 'UP');
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('touchstart', handleTouchStart);
    window.addEventListener('touchend', handleTouchEnd);

    return () => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('touchstart', handleTouchStart);
      window.removeEventListener('touchend', handleTouchEnd);
    };
  }, [onMove]);
};
```

### Task 3: Update `App.tsx`

**Files:**
- Modify: `src/App.tsx`

- [ ] **Step 1: Integrate hooks and update UI**

```tsx
import { useGame } from './hooks/useGame';
import { useMoveListeners } from './hooks/useMoveListeners';
import GameBoard from './components/GameBoard';
import './App.css';

function App() {
  const { state, move, restart, changeSize } = useGame(4);

  useMoveListeners(move);

  return (
    <div className="app">
      <header className="header">
        <div className="title-container">
          <h1 className="title">2048</h1>
          <div className="scores">
            <div className="score-box">
              <span className="label">SCORE</span>
              <span className="value">{state.score}</span>
            </div>
            <div className="score-box">
              <span className="label">BEST</span>
              <span className="value">{state.bestScore}</span>
            </div>
          </div>
        </div>
        <div className="controls">
          <div className="size-selector">
            {[3, 4, 5].map((s) => (
              <button
                key={s}
                className={`btn-size ${state.size === s ? 'active' : ''}`}
                onClick={() => changeSize(s)}
              >
                {s}x{s}
              </button>
            ))}
          </div>
          <button className="btn-restart" onClick={restart}>New Game</button>
        </div>
      </header>

      <main className="game-wrapper">
        <GameBoard tiles={state.tiles} size={state.size} />
        
        {state.status !== 'playing' && (
          <div className="game-message">
            <h2>{state.status === 'won' ? 'You Win!' : 'Game Over!'}</h2>
            <button className="btn-restart" onClick={restart}>Try Again</button>
          </div>
        )}
      </main>

      <footer className="footer">
        <p>Use arrow keys or swipe to move tiles.</p>
      </footer>
    </div>
  );
}

export default App;
```
